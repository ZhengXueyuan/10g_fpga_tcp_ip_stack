$ ... "sed -n '204,258p' \$L/cdev_sgdma.c"      (check_transfer_align)
static int check_transfer_align(engine, buf, count, pos, sync) {
        /* AXI ST or AXI MM non-incremental addressing mode? */
        if (engine->non_incr_addr) {
                int buf_lsb = (int)((uintptr_t)buf) & (engine->addr_align - 1);
                size_t len_lsb = count & ((size_t)engine->len_granularity - 1);
                int pos_lsb = (int)pos & (engine->addr_align - 1);
                if (buf_lsb != 0) return -EINVAL;
                if ((pos_lsb != 0) && (sync)) return -EINVAL;
                if (len_lsb != 0) return -EINVAL;
        } else {  /* AXI MM incremental addressing mode */
                int buf_lsb = (int)((uintptr_t)buf) & (engine->addr_align - 1);
                int pos_lsb = (int)pos & (engine->addr_align - 1);
                if (buf_lsb != pos_lsb) return -EINVAL;
        }
        return 0;
}

=> 【结论】单次 read() 大小 无 64 KB 级硬顶: 用户缓冲按页切成 sw_desc (每段 <= min(page, desc_blen_max)),
   每 2048 个描述符构成 1 个 transfer, 由 tfer[2] 乒乓 + while(nents) 链式续跑。
   内部缓冲 (ST C2H) = cyclic_result 回写数组 desc_max * sizeof(struct xdma_result) = 2048*32 B = 64 KB。
